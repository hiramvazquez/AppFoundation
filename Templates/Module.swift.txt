import AppFoundation
{{#newService}}
import CoreNetworking
{{/newService}}
{{#newStore}}
import SwiftData
{{/newStore}}
import Foundation

/// Registra el feature {{Feature}} en un `Container`: el composition root es el único
/// sitio del proyecto que conoce los tipos concretos detrás de sus protocolos (M4) —
/// ninguna Logic/Service/Store llama a `Container.shared`/`resolve(`/`@Inject` por su
/// cuenta (`ArchLint.R10`).
///
/// ```swift
{{#newService}}
{{#newStore}}
/// Container.shared.register(modules: [try! {{Feature}}Module(baseURL: URL(string: "https://api.example.com")!)])
{{/newStore}}
{{^newStore}}
/// Container.shared.register(modules: [{{Feature}}Module(baseURL: URL(string: "https://api.example.com")!)])
{{/newStore}}
{{/newService}}
{{^newService}}
{{#newStore}}
/// Container.shared.register(modules: [try! {{Feature}}Module()])
{{/newStore}}
{{^newStore}}
/// Container.shared.register(modules: [{{Feature}}Module()])
{{/newStore}}
{{/newService}}
/// // Vista raíz:
/// {{Feature}}View(viewModel: Container.shared.resolve())
/// ```
public struct {{Feature}}Module: DependencyModule {
{{#newService}}
    private let baseURL: URL
{{/newService}}
{{#newStore}}
    private let modelContainer: ModelContainer
{{/newStore}}

{{#newService}}
{{#newStore}}
    public init(baseURL: URL, modelContainer: ModelContainer? = nil) throws {
        self.baseURL = baseURL
        self.modelContainer = try modelContainer ?? ModelContainer(for: {{Feature}}Record.self)
    }
{{/newStore}}
{{^newStore}}
    public init(baseURL: URL) {
        self.baseURL = baseURL
    }
{{/newStore}}
{{/newService}}
{{^newService}}
{{#newStore}}
    public init(modelContainer: ModelContainer? = nil) throws {
        self.modelContainer = try modelContainer ?? ModelContainer(for: {{Feature}}Record.self)
    }
{{/newStore}}
{{^newStore}}
    public init() {}
{{/newStore}}
{{/newService}}

    public func register(in container: Container) {
{{#newService}}
        // MARK: Networking wiring
        container.register(APIServiceProtocol.self) { [baseURL] _ in
            APIService(configuration: NetworkingConfiguration(baseURL: baseURL))
        }

        // MARK: Service
        container.register({{Feature}}Servicing.self) { c in
            {{Feature}}Service(api: c.resolve())
        }
{{/newService}}
{{#serviceFrom}}
        // MARK: Service (reutilizado)
        // {{Feature}}Logic depende de 'any {{ServiceFeature}}Servicing' — {{ServiceFeature}}Module ya
        // lo registra, así que aquí no hay nada que registrar. Recuerda registrar también
        // {{ServiceFeature}}Module en el Container.
{{/serviceFrom}}
{{#noService}}
        // MARK: Service (pendiente)
        // TODO: registra {{Feature}}Servicing.self — {{Feature}}Logic no resuelve sin una
        // implementación concreta (o vuelve a generar con --service-from <FeatureExistente>
        // para reutilizar una real).
{{/noService}}
{{#newStore}}

        // MARK: Store
        container.register({{Feature}}Storing.self) { [modelContainer] _ in
            SwiftData{{Feature}}Store(modelContainer: modelContainer)
        }
{{/newStore}}
{{#storeFrom}}

        // MARK: Store (reutilizado)
        // {{Feature}}Logic depende de 'any {{StoreFeature}}Storing' — {{StoreFeature}}Module ya lo
        // registra, así que aquí no hay nada que registrar. Recuerda registrar también
        // {{StoreFeature}}Module en el Container.
{{/storeFrom}}
{{#noStore}}

        // MARK: Store (pendiente)
        // TODO: registra {{Feature}}Storing.self — {{Feature}}Logic no resuelve sin una
        // implementación concreta (o vuelve a generar con --store-from <FeatureExistente>
        // para reutilizar una real).
{{/noStore}}

        // MARK: Logic
{{#both}}
        container.register({{Feature}}LogicProtocol.self) { c in
            {{Feature}}Logic({{feature}}Service: c.resolve(), {{feature}}Store: c.resolve())
        }
{{/both}}
{{^both}}
{{#api}}
        container.register({{Feature}}LogicProtocol.self) { c in
            {{Feature}}Logic({{feature}}Service: c.resolve())
        }
{{/api}}
{{#local}}
        container.register({{Feature}}LogicProtocol.self) { c in
            {{Feature}}Logic({{feature}}Store: c.resolve())
        }
{{/local}}
{{#none}}
        container.register({{Feature}}LogicProtocol.self) { _ in {{Feature}}Logic() }
{{/none}}
{{/both}}

        // MARK: ViewModel — resuelto por la vista raíz, nunca construido por Logic/Service/Store.
        container.register({{Feature}}ViewModel.self, lifecycle: .transient) { c in
            {{Feature}}ViewModel(logic: c.resolve())
        }
    }
}

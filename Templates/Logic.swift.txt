///
/// `nonisolated` (M5): no depende del actor principal — sus métodos `async` corren en el
/// actor de quien la llama.
public nonisolated final class {{Feature}}Logic: {{Feature}}LogicProtocol {
{{#api}}
    private let {{feature}}Service: any {{ServicingType}}
{{/api}}
{{#local}}
    private let {{feature}}Store: any {{StoringType}}
{{/local}}

{{#both}}
    /// - Parameters:
    ///   - {{feature}}Service: La llamada a red de este feature. Inyectada como protocolo.
    ///   - {{feature}}Store: La caché local. Inyectada como protocolo — nunca el
    ///     `SwiftData{{Feature}}Store`/`InMemory{{Feature}}Store` concreto
    ///.
    public init({{feature}}Service: any {{ServicingType}}, {{feature}}Store: any {{StoringType}}) {
        self.{{feature}}Service = {{feature}}Service
        self.{{feature}}Store = {{feature}}Store
    }

    public func cached() async -> [{{Feature}}Item] {
{{#storeFrom}}
        // {{feature}}Store is `any {{StoringType}}` (--store-from {{StoreFeature}}): it returns
        // `[{{StoreFeature}}Item]`, mapped field-by-field into this feature's own
        // `{{Feature}}Item` — every domain item this generator produces is `{ id, title }`-
        // shaped, so this always compiles; widen it by hand once either type grows real
        // fields of its own.
        (try? await {{feature}}Store.fetchAll())?.map { {{Feature}}Item(id: $0.id, title: $0.title) } ?? []
{{/storeFrom}}
{{^storeFrom}}
        (try? await {{feature}}Store.fetchAll()) ?? []
{{/storeFrom}}
    }

    public func refresh() async throws -> [{{Feature}}Item] {
        do {
{{#serviceFrom}}
            // {{feature}}Service is `any {{ServicingType}}` (--service-from {{ServiceFeature}}):
            // same `{ id, title }` mapping as cached() above, on the way in this time.
            let items = try await {{feature}}Service.fetchItems().map { {{Feature}}Item(id: $0.id, title: $0.title) }
{{/serviceFrom}}
{{^serviceFrom}}
            let items = try await {{feature}}Service.fetchItems()
{{/serviceFrom}}
            // Best-effort: los items frescos se devuelven aunque falle persistirlos — un
            // fallo de caché no debe convertir un refresh exitoso en un error.
{{#storeFrom}}
            try? await {{feature}}Store.replaceAll(items.map { {{StoreFeature}}Item(id: $0.id, title: $0.title) })
{{/storeFrom}}
{{^storeFrom}}
            try? await {{feature}}Store.replaceAll(items)
{{/storeFrom}}
            return items
        } catch {
            throw Self.mapError(error)
        }
    }

    private static func mapError(_ error: APIError) -> {{Feature}}Error {
        switch error.category {
        case .offline: return .offline
        case .server: return .server
        default: return .unknown
        }
    }
{{/both}}
{{^both}}
{{#api}}
    /// - Parameter {{feature}}Service: La única llamada a red de este feature. Inyectada
    ///   como protocolo — nunca `{{Feature}}Service` directamente
    ///.
    public init({{feature}}Service: any {{ServicingType}}) {
        self.{{feature}}Service = {{feature}}Service
    }

    public func load() async throws -> [{{Feature}}Item] {
        do {
{{#serviceFrom}}
            // {{feature}}Service is `any {{ServicingType}}` (--service-from {{ServiceFeature}}): it
            // returns `[{{ServiceFeature}}Item]`, mapped field-by-field into this feature's own
            // `{{Feature}}Item` — every domain item this generator produces is `{ id, title }`-
            // shaped, so this always compiles; widen it by hand once either type grows real
            // fields of its own.
            return try await {{feature}}Service.fetchItems().map { {{Feature}}Item(id: $0.id, title: $0.title) }
{{/serviceFrom}}
{{^serviceFrom}}
            return try await {{feature}}Service.fetchItems()
{{/serviceFrom}}
        } catch {
            throw Self.mapError(error)
        }
    }

    private static func mapError(_ error: APIError) -> {{Feature}}Error {
        switch error.category {
        case .offline: return .offline
        case .server: return .server
        default: return .unknown
        }
    }
{{/api}}
{{#local}}
    /// - Parameter {{feature}}Store: La única dependencia de este feature. Inyectada como
    ///   protocolo — nunca `SwiftData{{Feature}}Store`/`InMemory{{Feature}}Store`
    ///   directamente.
    public init({{feature}}Store: any {{StoringType}}) {
        self.{{feature}}Store = {{feature}}Store
    }

    public func load() async throws -> [{{Feature}}Item] {
        do {
{{#storeFrom}}
            // {{feature}}Store is `any {{StoringType}}` (--store-from {{StoreFeature}}): same
            // `{ id, title }` mapping as the api-only variant does for its Service.
            return try await {{feature}}Store.fetchAll().map { {{Feature}}Item(id: $0.id, title: $0.title) }
{{/storeFrom}}
{{^storeFrom}}
            return try await {{feature}}Store.fetchAll()
{{/storeFrom}}
        } catch {
            throw {{Feature}}Error.storageFailure
        }
    }
{{/local}}
{{#none}}
    /// La variante "sin datos": `Logic` no depende de nada — ni `*Servicing` ni
    /// `*Storing` — pero sigue siendo su propio tipo: la regla de negocio (aquí, "Hola
    /// mundo") vive aquí y en ningún otro sitio, exactamente igual que en las otras tres
    /// variantes.
    public init() {}

    public func load() async throws -> [{{Feature}}Item] {
        [{{Feature}}Item(title: "Hola mundo")]
    }
{{/none}}
{{/both}}
}
